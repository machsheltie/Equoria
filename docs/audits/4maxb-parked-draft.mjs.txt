/**
 * bareDefaultGuardZeroCorruption.sentinel.test.mjs
 *
 * Sentinel-positive test for the bare-default-guard corruption defect (Equoria-4maxb).
 */

import { describe, it, expect, beforeAll, afterAll } from '@jest/globals';
import { randomBytes } from 'node:crypto';
import prisma from '../../packages/database/prismaClient.mjs';
import { discoverTraitsForHorse } from '../utils/traitDiscovery.mjs';
import { fixtureColor } from '../tests/helpers/fixtureColor.mjs';

let ownerId;
const createdHorseIds = [];

async function createTestHorse(overrides = {}) {
  const horse = await prisma.horse.create({
    data: {
      ...fixtureColor(),
      name: ,
      sex: 'Mare',
      dateOfBirth: new Date(),
      age: 5,
      userId: ownerId,
      bondScore: 0,
      stressLevel: 0,
      ...overrides,
    },
  });
  createdHorseIds.push(horse.id);
  return horse;
}

beforeAll(async () => {
  const uniq = randomBytes(8).toString('hex');
  const owner = await prisma.user.create({
    data: {
      email: ,
      username: ,
      password: 'irrelevant-hash',
      firstName: 'Zero',
      lastName: 'Tester',
      money: 1000,
    },
  });
  ownerId = owner.id;
}, 120000);

afterAll(async () => {
  if (createdHorseIds.length > 0) {
    await prisma.horse.deleteMany({ where: { id: { in: createdHorseIds } } });
  }
  if (ownerId) {
    await prisma.user.deleteMany({ where: { id: ownerId } });
  }
}, 120000);

describe('SENTINEL: Bare-default guards corrupt 0-values (Equoria-4maxb)', () => {
  it('calm horse (stressLevel=0) should NOT be treated as max-stressed (100)', async () => {
    const horse = await createTestHorse({
      stressLevel: 0,
      bondScore: 50,
    });
    const fetched = await prisma.horse.findUnique({
      where: { id: horse.id },
      select: { stressLevel: true },
    });
    expect(fetched.stressLevel).toBe(0);
    const result = await discoverTraitsForHorse(horse.id);
    expect(result).toBeDefined();
    expect(result.success).toBe(true);
  }, 120000);

  it('unbonded horse (bondScore=0) should NOT be read as neutral-bonded (50)', async () => {
    const horse = await createTestHorse({
      bondScore: 0,
      stressLevel: 50,
    });
    const fetched = await prisma.horse.findUnique({
      where: { id: horse.id },
      select: { bondScore: true },
    });
    expect(fetched.bondScore).toBe(0);
    const { applyBondingChange } = await import('../utils/bondingModifiers.mjs');
    const result = applyBondingChange(horse, 'grooming', { duration: 30 });
    expect(result.success).toBe(true);
    expect(result.oldBondScore).toBe(0);
    expect(result.newBondScore).toBeGreaterThan(0);
  }, 120000);

  it('normal bondScore/stressLevel values work correctly', async () => {
    const horse = await createTestHorse({
      bondScore: 50,
      stressLevel: 50,
    });
    const fetched = await prisma.horse.findUnique({
      where: { id: horse.id },
      select: { bondScore: true, stressLevel: true },
    });
    expect(fetched.bondScore).toBe(50);
    expect(fetched.stressLevel).toBe(50);
    const result = await discoverTraitsForHorse(horse.id);
    expect(result.success).toBe(true);
  }, 120000);
});
